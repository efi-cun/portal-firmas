    function setFieldSafe(form, fieldName, value) {

                field.setText(value != null ? String(value) : '');

    // Helper para limpiar textos dummies de placeholders
    function cleanDummyText(form, fieldName, dummyKeywords) {
        try {
            const field = form.getField(fieldName);
            if (field && typeof field.getText === 'function') {
                const val = (field.getText() || '').trim().toLowerCase();
                if (dummyKeywords.some(w => val === w || val.startsWith(w) || val.includes(w))) {

    // Helper para ubicar widget y página
    function getWidgetTarget(form, pages, fieldName, fallbackPageIdx = 0) {
        try {
            const field = form.getField(fieldName);
            if (field) {
                const widgets = field.acroField.getWidgets();
                if (widgets && widgets.length > 0) {
                    const widget = widgets[0];
                    const rect = widget.getRectangle();
                    let widgetPage = null;
                    try {
                        const pageRef = widget.P();
                        if (pageRef) {
                            for (let i = 0; i < pages.length; i++) {
                                if (pages[i].ref.toString() === pageRef.toString()) {
                                    widgetPage = pages[i];
                                    break;
                                }
                            }
                        }
                    } catch (errP) {}
                    if (!widgetPage) {
                        const pMatch = fieldName.match(/_p(\d+)$/i);
                        if (pMatch) {
                            const pIdx = parseInt(pMatch[1], 10) - 1;
                            if (pIdx >= 0 && pIdx < pages.length) {
                                widgetPage = pages[pIdx];
                            }
                        }
                    }
                    if (!widgetPage && fallbackPageIdx >= 0 && fallbackPageIdx < pages.length) {
                        widgetPage = pages[fallbackPageIdx];
                    }
                    return {
                        page: widgetPage,
                        x: rect.x,
                        y: rect.y,
                        width: rect.width || 140,
                        height: rect.height || 40
                    };
                }
            }
        } catch (e) {}
        return null;
    }

    // Función que procesa e inyecta los datos del evaluador/jefe en el PDF
    async function prepararPDFPorAdmin(file) {
        const arrayBuffer = await file.arrayBuffer();
        const pdfDocLib = await PDFLib.PDFDocument.load(arrayBuffer, { ignoreEncryption: true });
        const form = pdfDocLib.getForm();
        const pages = pdfDocLib.getPages();

        // 1. Datos del Jefe Inmediato (Hoja 4)
        const jefeNombre = document.getElementById('admin-jefe-nombre').value.trim();
        const jefeCedula = document.getElementById('admin-jefe-cedula').value.trim();
        const jefeCargo = document.getElementById('admin-jefe-cargo').value.trim();
        setFieldSafe(form, 'Eval_Jefe_Inmediato_p4', jefeNombre);
        setFieldSafe(form, 'Eval_DNI_Jefe_p4', jefeCedula);
        setFieldSafe(form, 'Eval_Cargo_Jefe_p4', jefeCargo);

        // 2. Datos de Contrato y Período de Prueba (Hoja 4)
        const tipoContrato = document.getElementById('admin-tipo-contrato').value.trim();
        let fechaInicioPrueba = document.getElementById('admin-fecha-inicio-prueba').value;
        if (fechaInicioPrueba && fechaInicioPrueba.includes('-')) {
            const p = fechaInicioPrueba.split('-');
            if (p.length === 3) fechaInicioPrueba = `${p[2]}/${p[1]}/${p[0]}`;
        }
        let fechaFinPrueba = document.getElementById('admin-fecha-fin-prueba').value;
        if (fechaFinPrueba && fechaFinPrueba.includes('-')) {
            const p = fechaFinPrueba.split('-');
            if (p.length === 3) fechaFinPrueba = `${p[2]}/${p[1]}/${p[0]}`;
        }
        const periodoPrueba = document.getElementById('admin-periodo-prueba').value.trim();

        setFieldSafe(form, 'Eval_Tipo_Contrato_p4', tipoContrato);
        setFieldSafe(form, 'Eval_Fecha_Inicio_Prueba_p4', fechaInicioPrueba);
        setFieldSafe(form, 'Eval_Fecha_Fin_Prueba_p4', fechaFinPrueba);
        setFieldSafe(form, 'Eval_Tipo_Evaluacion_p4', periodoPrueba);

        // 3. Datos del Evaluador (Hoja 5)
        const evaluadorNombre = document.getElementById('admin-evaluador-nombre').value.trim();
        const evaluadorCargo = document.getElementById('admin-evaluador-cargo').value.trim();
        setFieldSafe(form, 'Nombre_Evaluador_p5', evaluadorNombre);
        setFieldSafe(form, 'Cargo_Evaluador_p5', evaluadorCargo);

        // 4. Campos Opcionales (Fortalezas, Recomendaciones, Compromisos, Observaciones)
        const fortalezas = document.getElementById('admin-fortalezas').value.trim();
        const recomendaciones = document.getElementById('admin-recomendaciones').value.trim();
        const observaciones = document.getElementById('admin-observaciones').value.trim();
        setFieldSafe(form, 'Fortalezas_p5', fortalezas);
        setFieldSafe(form, 'Recomendaciones_p5', recomendaciones);
        setFieldSafe(form, 'Observaciones_Capital_Social_p5', observaciones);

        for (let i = 1; i <= 5; i++) {
            const desc = (document.getElementById(`admin-comp-desc-${i}`)?.value || '').trim();
            const res = (document.getElementById(`admin-comp-res-${i}`)?.value || '').trim();
            const seg = (document.getElementById(`admin-comp-seg-${i}`)?.value || '').trim();
            setFieldSafe(form, `Compromiso_${i}_Desc_p5`, desc);
            setFieldSafe(form, `Compromiso_${i}_Resultado_p5`, res);
            setFieldSafe(form, `Compromiso_${i}_Seguimiento_p5`, seg);
        }

        // 5. Limpieza de placeholders antiguos de firma/cédula/cargo para quien recibe
        cleanDummyText(form, 'Firma_Digital_Recibi_p1', ['firma', 'quien recibe']);
        cleanDummyText(form, 'Firma_Digital_Recibi_p2', ['firma', 'quien recibe']);
        cleanDummyText(form, 'Firma_Digital_Recibi_p3', ['firma', 'quien recibe']);
        cleanDummyText(form, 'Firma_Evaluado_p5', ['firma', 'quien recibe']);
        cleanDummyText(form, 'Firma_Capital_Social_p5', ['firma', 'quien recibe']);
        cleanDummyText(form, 'Firma_Evaluador_p5', ['firma', 'evaluador']);
        cleanDummyText(form, 'Cedula_Recibi_p1', ['cedula', 'cédula', 'quien recibe']);
        cleanDummyText(form, 'Cedula_Recibi_p2', ['cedula', 'cédula', 'quien recibe']);
        cleanDummyText(form, 'Cedula_Recibi_p3', ['cedula', 'cédula', 'quien recibe']);
        cleanDummyText(form, 'Eval_Cedula_Evaluado_p4', ['cedula', 'cédula', 'quien recibe']);
        cleanDummyText(form, 'Eval_Nombre_Evaluado_p4', ['nombre quien recibe']);
        cleanDummyText(form, 'Nombre_Evaluado_p5', ['nombre quien recibe']);
        cleanDummyText(form, 'Nombre_Capital_Social_p5', ['nombre  de quien recibe', 'nombre de quien recibe']);
        cleanDummyText(form, 'Cargo_Evaluado_p5', ['cargo quien recibe']);
        cleanDummyText(form, 'Cargo_Capital_Social_p5', ['cargo quien recibe']);
        cleanDummyText(form, 'Eval_Cargo_Evaluado_p4', ['cargo quien recibe']);
        cleanDummyText(form, 'Comentarios_Evaluado_p5', ['comentarios']);

        // 6. Firma del Evaluador (si se dibujó en el canvas del administrador)
        if (adminSignaturePad && !adminSignaturePad.isEmpty()) {
            const evalSigDataUrl = adminSignaturePad.toDataURL('image/png');
            const evalSigImage = await pdfDocLib.embedPng(evalSigDataUrl);
            const target = getWidgetTarget(form, pages, 'Firma_Evaluador_p5', pages.length - 1);
            
            // Limpiar texto en casilla de firma
            setFieldSafe(form, 'Firma_Evaluador_p5', '');

            if (target) {
                target.page.drawImage(evalSigImage, {
                    x: target.x + 2,
                    y: target.y + 2,
                    width: Math.max(target.width - 4, 10),
                    height: Math.max(target.height - 4, 10)
                });
            } else {
                // Posición de respaldo en hoja 5 (casilla del evaluador)
                const p5 = pages[pages.length - 1];
                p5.drawImage(evalSigImage, { x: 127, y: 333.5, width: 219, height: 18.5 });
            }
        }

        // Guardamos SIN aplanar para que los campos del empleado continúen interactivos
        return await pdfDocLib.save();
    }

    function descargarYMostrarPDF(id) {
        const statusDiv = document.getElementById('signer-status');
        statusDiv.className = 'status-loading'; 
        statusDiv.style.display = 'block';
        statusDiv.textContent = "Descargando documento seguro para firma...";

        fetch(GOOGLE_APP_URL, {
            method: 'POST',
            body: JSON.stringify({ action: "obtener_pdf", docId: id }),
            headers: { "Content-Type": "text/plain;charset=utf-8" }
        })
        .then(res => res.json())
        .then(data => {
            if(data.status === 'success' && data.base64) {
                statusDiv.style.display = 'none';
                const binaryString = window.atob(data.base64);
                pdfBytes = new Uint8Array(binaryString.length);
                for (let i = 0; i < binaryString.length; i++) {
                    pdfBytes[i] = binaryString.charCodeAt(i);
                }
                mostrarPDFEnPantalla(pdfBytes);
                preCargarDatosFirmante(pdfBytes);
            } else {
                statusDiv.className = 'status-error';
                statusDiv.textContent = "Error al descargar el PDF. Verifica el enlace.";
            }
        })
        .catch(err => {
            statusDiv.className = 'status-error';
            statusDiv.textContent = "Error de red al descargar el documento.";
        });
    }

    function mostrarPDFEnPantalla(bytes) {
        pdfjsLib.getDocument(bytes).promise.then(pdf => {
            pdfDoc = pdf;
            document.getElementById('page-count').textContent = pdf.numPages;
            pageNum = 1;
            renderPage(pageNum);
            document.getElementById('pdf-container').style.display = 'flex';
            document.getElementById('signature-section').style.display = 'flex';
            document.getElementById('page-info').style.display = 'block';

                    signerPadCanvas.width = signerPadCanvas.offsetWidth * ratio || 520;
                    signerPadCanvas.height = signerPadCanvas.offsetHeight * ratio || 180;
                    signerPadCanvas.getContext("2d").scale(ratio, ratio);

            }, 100);
        });

    function renderPage(num) {
        pdfDoc.getPage(num).then(page => {
            const viewport = page.getViewport({ scale: 1.8 });
            canvas.height = viewport.height;
            canvas.width = viewport.width;
            page.render({ canvasContext: ctx, viewport: viewport });
        });
        document.getElementById('page-num').textContent = num;
        document.getElementById('prev-page').disabled = num <= 1;
        document.getElementById('next-page').disabled = num >= pdfDoc.numPages;
    }

    // Función que procesa la firma del empleado en las casillas correspondientes
    async function firmarDocumentoEmpleado() {
        const nombre = document.getElementById('nombre').value.trim();
        const cedula = document.getElementById('cedula').value.trim();
        const cargo = document.getElementById('cargo').value.trim();
        const fecha = document.getElementById('fecha').value.trim();
        const comentarios = (document.getElementById('comentarios')?.value || '').trim();

        if (!nombre || !cedula || !cargo || !fecha) {
            alert('Por favor completa todos tus datos: Nombre, Cédula, Cargo y Fecha.');
            return null;
        }
        if (!signaturePad || signaturePad.isEmpty()) {
            alert('Por favor dibuja tu firma en el recuadro.');
            return null;
        }

        const pdfDocLib = await PDFLib.PDFDocument.load(pdfBytes, { ignoreEncryption: true });
        const signatureImageBytes = signaturePad.toDataURL('image/png');
        const signatureImage = await pdfDocLib.embedPng(signatureImageBytes);
        const pages = pdfDocLib.getPages();

        let fechaDisplay = fecha;
        if (fecha && fecha.includes('-')) {
            const parts = fecha.split('-');
            if (parts.length === 3) fechaDisplay = `${parts[2]}/${parts[1]}/${parts[0]}`;
        }

        const targetsToSign = [];
        let firmaColocada = false;

        try {
            const form = pdfDocLib.getForm();
            const fields = form.getFields();

            fields.forEach(field => {
                const rawName = field.getName();
                const name = rawName.toLowerCase();

                // Excluir de forma estricta los campos del Evaluador y Jefe
                const isEvaluador = name.includes('evaluador');
                const isJefe = name.includes('jefe');
                const isSignatureField = name.includes('firma');

                // 1. CASILLAS DE FIRMA: Únicamente campos que incluyan 'firma'
                if (isSignatureField) {
                    // Firmas del destinatario (empleado):
                    // - Hojas 1, 2, 3: Firma_Digital_Recibi_p1, p2, p3
                    // - Hoja 5: Firma_Evaluado_p5 (Evaluado)
                    // - Hoja 5: Firma_Capital_Social_p5 (Recibido por en Espacio de Capital Social)
                    // Se excluye estrictamente Firma_Evaluador_p5
                    const isRecipientSig = !isEvaluador && (
                        name.includes('recibi') || 
                        name.includes('recibí') || 
                        name.includes('evaluado') || 
                        name.includes('capital') ||
                        name.includes('digital')
                    );

                    if (isRecipientSig) {
                        try {
                            if (typeof field.setText === 'function') {
                                field.setText(''); // Limpiar texto de placeholder como 'firma' o 'firma quien recibe'
                            }
                            const widgets = field.acroField.getWidgets();
                            if (widgets && widgets.length > 0) {
                                const widget = widgets[0];
                                const rect = widget.getRectangle();
                                
                                let widgetPage = null;
                                try {
                                    const pageRef = widget.P();
                                    if (pageRef) {
                                        for (let i = 0; i < pages.length; i++) {
                                            if (pages[i].ref.toString() === pageRef.toString()) {
                                                widgetPage = pages[i];
                                                break;
                                            }
                                        }
                                    }
                                } catch (errP) {}

                                if (!widgetPage) {
                                    const pMatch = rawName.match(/_p(\d+)$/i);
                                    if (pMatch) {
                                        const pIdx = parseInt(pMatch[1], 10) - 1;
                                        if (pIdx >= 0 && pIdx < pages.length) {
                                            widgetPage = pages[pIdx];
                                        }
                                    }
                                }

                                if (!widgetPage) {
                                    widgetPage = pages[0];
                                }

                                targetsToSign.push({
                                    page: widgetPage,
                                    x: rect.x,
                                    y: rect.y,
                                    width: rect.width || 140,
                                    height: rect.height || 40,
                                    fieldName: rawName
                                });
                                firmaColocada = true;
                            }
                        } catch(e) {}
                    }
                }
                // 2. CAMPOS DE TEXTO: (Cédula, Cargo, Nombre, Fecha, Comentarios)
                // Se asignan únicamente si NO es firma y NO pertenece al Evaluador ni al Jefe
                else if (typeof field.setText === 'function') {
                    if (!isEvaluador && !isJefe) {
                        // Cédula en Hojas 1, 2, 3 (Cedula_Recibi_p1..3) y Hoja 4 (Eval_Cedula_Evaluado_p4)
                        if (name.includes('cedula') || name.includes('cédula') || name.includes('documento') || name.includes('dni')) {
                            field.setText(cedula);
                        }
                        // Cargo en Hoja 4 (Eval_Cargo_Evaluado_p4), Hoja 5 (Cargo_Evaluado_p5) y Hoja 5 (Cargo_Capital_Social_p5)
                        else if (name.includes('cargo')) {
                            field.setText(cargo);
                        }
                        // Comentarios del Evaluado (Hoja 5)
                        else if (name.includes('comentarios')) {
                            field.setText(comentarios);
                        }
                        // Nombre en Hoja 4 (Eval_Nombre_Evaluado_p4), Hoja 5 (Nombre_Evaluado_p5) y Hoja 5 (Nombre_Capital_Social_p5)
                        else if (name.includes('nombre')) {
                            field.setText(nombre);
                        }
                        // Fecha en Hojas 1, 2, 3 (Fecha_Recibi_p1..3) y Hoja 5 (Fecha_Capital_Social_p5)
                        else if (name.includes('fecha') || name.includes('date')) {
                            if (name.includes('recibi') || name.includes('capital')) {
                                field.setText(fechaDisplay);
                            }
                        }
                    }
                }
            });

            // Aplanar formulario para sellar permanentemente todos los campos de texto
            form.flatten();

            // Estampar la firma del empleado en cada casilla objetivo identificada (con 2pt de padding para que encaje perfecto)
            if (firmaColocada && targetsToSign.length > 0) {
                targetsToSign.forEach(target => {
                    const padX = 2;
                    const padY = 2;
                    target.page.drawImage(signatureImage, {
                        x: target.x + padX,
                        y: target.y + padY,
                        width: Math.max(target.width - (padX * 2), 10),
                        height: Math.max(target.height - (padY * 2), 10)
                    });
                });
            } else {
                pages[0].drawImage(signatureImage, { x: 50, y: 50, width: 200, height: 80 });
            }
        } catch (formErr) {
            console.log("Aviso al procesar formulario:", formErr);
        }

        return await pdfDocLib.save();
    }

                body: JSON.stringify({
                    action: "guardar_firma",
                    pdfBase64: base64PDF,
                    fileName: fileName,
                    docId: docId
                }),
                headers: { "Content-Type": "text/plain;charset=utf-8" }
            })
            .then(res => res.json())
            .then(data => {
                if(data.status === 'success') {
                    statusDiv.className = 'status-success';
                    statusDiv.innerHTML = "🎉 <strong>¡Gracias!</strong> Tu documento ha sido firmado exitosamente y guardado en la empresa.";
                    document.getElementById('pdf-container').style.display = 'none';
                    document.getElementById('signature-section').style.display = 'none';
                    document.getElementById('page-info').style.display = 'none';
                } else {
                    statusDiv.className = 'status-error';
                    statusDiv.textContent = "Error al guardar el PDF en el servidor.";
                }
            })
            .catch(err => {
                statusDiv.className = 'status-error';
                statusDiv.textContent = "Error de red al enviar el documento firmado.";
            });
        } catch (error) {
            statusDiv.className = 'status-error';
            statusDiv.textContent = 'Error al generar el PDF: ' + error.message;
        }
    });

    // Botón: Descargar Copia Firmada Localmente
    document.getElementById('btn-descargar-firmante').addEventListener('click', async () => {
        try {
            const pdfBytesModified = await firmarDocumentoEmpleado();
            if (!pdfBytesModified) return;
            const blob = new Blob([pdfBytesModified], { type: 'application/pdf' });
            downloadBlob(blob, `FIRMADO_${fileName}`);
        } catch(e) {
            alert("Error al generar copia local: " + e.message);
        }
    });
</script>
